# MVP track dependencies - the eight packages docs/mvp/ actually uses, or
# seven once you have confirmed the API carries attendance and dropped lxml.
#
#   pip install -r requirements-mvp.txt
#
# Everything between the API and Tableau is Python. Nothing here listens on a
# port. The FootyStats key is the one secret, it lives in .env, and the
# pipeline runs without it once the archive is populated.
#
# Move to requirements.txt when you start the full track - it adds nothing you
# have to unlearn, only things the MVP does not reach for yet.

# Ingest. The FootyStats API is JSON over HTTP, so requests is the whole
# client; python-dotenv reads the paid API key out of .env so it never lands in
# source. Leave the key empty and the pipeline runs from data/raw_archive/.
requests>=2.31
python-dotenv>=1.0

# Only needed if the attendance fallback scraper turns out to be necessary -
# see the open question in docs/phases/00-data-access-and-the-clock.md. It is
# pandas.read_html's parser, not a direct import of ours. Delete this line once
# you have confirmed the API carries per-match attendance.
lxml>=5.0

# Store and transform. The SQL layer runs inside DuckDB; Python only reads the
# .sql file and hands it over.
duckdb>=1.0
pandas>=2.2

# Model. The one genuinely heavy install in the project - xgboost ships a
# compiled wheel. Unavoidable: the two-model comparison is the headline question.
xgboost>=2.0
scikit-learn>=1.4
numpy>=1.26
